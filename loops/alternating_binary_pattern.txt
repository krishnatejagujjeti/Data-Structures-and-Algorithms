
/*
print below star pattern
1
10
101
1010
10101


*/
function starPattern(n) {

    for (let i = 1; i <= n; i++) {
        let row = "";
        for (let j = 1; j <= i; j++) {
            if (j % 2 == 0)
                row = row + '0';
            else
                row = row + '1';

        }

        console.log(row);
    }

}
starPattern(8);
